CREATE TABLE companies (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    tax_id VARCHAR(50) NOT NULL UNIQUE,
    tax_id_password VARCHAR(50) NOT NULL,
    social_security_id VARCHAR(50) NOT NULL UNIQUE,
    social_security_id_password VARCHAR(50) NOT NULL,
    address VARCHAR(255),
    phone VARCHAR(50),
    email VARCHAR(255)
);

CREATE TABLE employees (
    id BIGSERIAL PRIMARY KEY,
    company_id BIGINT NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    tax_id VARCHAR(50) NOT NULL UNIQUE,
    hire_date DATE NOT NULL,
    base_salary DECIMAL(12,2) NOT NULL,
    social_security_contribution DECIMAL(12,2) NOT NULL,
    tax_id_contribution DECIMAL(12,2) NOT NULL,
    total_contributions DECIMAL(12,2) NOT NULL,
    net_salary DECIMAL(12,2) NOT NULL,
    dependents DECIMAL(12,2) NOT NULL,
    position VARCHAR(100),
    department VARCHAR(100),
    bank_account VARCHAR(50),
    email VARCHAR(255),
    phone VARCHAR(50),
    FOREIGN KEY (company_id) REFERENCES companies(id)
);

CREATE TABLE salary_receipts (
    id BIGSERIAL PRIMARY KEY,
    employee_id BIGINT NOT NULL,
    payment_date DATE NOT NULL,
    month VARCHAR(20) NOT NULL,
    year INTEGER NOT NULL,
    base_salary DECIMAL(12,2) NOT NULL,
    social_security_contribution DECIMAL(12,2) NOT NULL,
    tax_id_contribution DECIMAL(12,2) NOT NULL,
    total_contributions DECIMAL(12,2) NOT NULL,
    overtime DECIMAL(12,2),
    bonus DECIMAL(12,2),
    meal_allowance DECIMAL(12,2),
    transport_allowance DECIMAL(12,2),
    other_allowances DECIMAL(12,2),
    income_tax DECIMAL(12,2) NOT NULL,
    gross_amount DECIMAL(12,2) NOT NULL,
    net_amount DECIMAL(12,2) NOT NULL,
    notes TEXT,
    FOREIGN KEY (employee_id) REFERENCES employees(id)
);
